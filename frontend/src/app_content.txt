import { useState } from "react";
import { analyzePhishingEmail } from "./api/phishing.js";
import { analyzeURL } from "./api/urlRisk.js";

export default function App() {
  const [sender, setSender] = useState("");
  const [subject, setSubject] = useState("");
  const [body, setBody] = useState("");

  const [phishingResult, setPhishingResult] = useState(null);
  const [phishingLoading, setPhishingLoading] = useState(false);
  const [phishingError, setPhishingError] = useState("");

  const [url, setUrl] = useState("");
  const [urlResult, setUrlResult] = useState(null);
  const [urlLoading, setUrlLoading] = useState(false);
  const [urlError, setUrlError] = useState("");

  async function handleAnalyzeEmail() {
    setPhishingError("");
    setPhishingResult(null);

    if (!subject.trim() && !body.trim()) {
      setPhishingError("Please enter an email subject or body.");
      return;
    }

    setPhishingLoading(true);

    try {
      const data = await analyzePhishingEmail({
        sender,
        subject,
        body,
      });

      setPhishingResult(data);
    } catch (err) {
      setPhishingError(err.message || "Unable to analyze the email.");
    } finally {
      setPhishingLoading(false);
    }
  }

  async function handleAnalyzeURL() {
    setUrlError("");
    setUrlResult(null);

    if (!url.trim()) {
      setUrlError("Please enter a URL.");
      return;
    }

    setUrlLoading(true);

    try {
      const data = await analyzeURL(url);
      setUrlResult(data);
    } catch (err) {
      setUrlError(err.message || "Unable to analyze the URL.");
    } finally {
      setUrlLoading(false);
    }
  }

  const phishingProbability = phishingResult
    ? (phishingResult.probability_phishing * 100).toFixed(2)
    : null;

  return (
    <div className="page">
      <header className="topbar">
        <div>
          <p className="eyebrow">
            Cyber AI Hackathon 2026 � University of Derby
          </p>
          <h1>TrustGuard AI</h1>
          <p className="tagline">
            Adaptive AI for Cyber Threat Detection & Digital Trust
          </p>
        </div>

        <div className="badge badge--ok">
          <span className="badge__dot" />
          Cyber Threat Detection
        </div>
      </header>

      <main className="grid">
        <section className="panel">
          <h2>Phishing Email Analysis</h2>

          <p className="lede">
            Analyze an email using the TrustGuard AI phishing detection model.
          </p>

          <label>
            Sender
            <input
              type="text"
              value={sender}
              onChange={(e) => setSender(e.target.value)}
              placeholder="security@example.com"
            />
          </label>

          <label>
            Subject
            <input
              type="text"
              value={subject}
              onChange={(e) => setSubject(e.target.value)}
              placeholder="Urgent account verification required"
            />
          </label>

          <label>
            Email Body
            <textarea
              rows="7"
              value={body}
              onChange={(e) => setBody(e.target.value)}
              placeholder="Paste the email content here..."
            />
          </label>

          <button onClick={handleAnalyzeEmail} disabled={phishingLoading}>
            {phishingLoading ? "Analyzing..." : "Analyze Email"}
          </button>

          {phishingError && (
            <div className="callout callout--warn">
              <strong>Error</strong>
              <p>{phishingError}</p>
            </div>
          )}
        </section>

        <section className="panel">
          <h2>Phishing Result</h2>

          {!phishingResult && !phishingLoading && (
            <p className="lede">
              Submit an email to see the AI detection result.
            </p>
          )}

          {phishingLoading && (
            <p className="lede">TrustGuard AI is analyzing the email...</p>
          )}

          {phishingResult && (
            <>
              <div className="result">
                <p className="eyebrow">Prediction</p>
                <h3>{phishingResult.prediction.toUpperCase()}</h3>

                <p>
                  Phishing probability:{" "}
                  <strong>{phishingProbability}%</strong>
                </p>

                <p>
                  Threat status:{" "}
                  <strong>{phishingResult.threat_status}</strong>
                </p>
              </div>

              <div className="meta">
                <div>
                  <strong>Model</strong>
                  <span>{phishingResult.model}</span>
                </div>

                <div>
                  <strong>Version</strong>
                  <span>{phishingResult.model_version}</span>
                </div>
              </div>

              <h3>Top AI Indicators</h3>

              <ul className="module-list">
                {phishingResult.explanation.map((item) => (
                  <li key={item.feature}>
                    <span>{item.feature}</span>
                    <span className="chip">
                      {item.contribution_to_phishing.toFixed(3)}
                    </span>
                  </li>
                ))}
              </ul>
            </>
          )}
        </section>

        <section className="panel">
          <h2>URL Risk Analysis</h2>

          <p className="lede">
            Analyze a URL for suspicious patterns and basic security risks.
          </p>

          <label>
            URL
            <input
              type="text"
              value={url}
              onChange={(e) => setUrl(e.target.value)}
              placeholder="https://example.com/login"
            />
          </label>

          <button onClick={handleAnalyzeURL} disabled={urlLoading}>
            {urlLoading ? "Analyzing..." : "Analyze URL"}
          </button>

          {urlError && (
            <div className="callout callout--warn">
              <strong>Error</strong>
              <p>{urlError}</p>
            </div>
          )}
        </section>

        <section className="panel">
          <h2>URL Analysis Result</h2>

          {!urlResult && !urlLoading && (
            <p className="lede">
              Enter a URL and click <strong>Analyze URL</strong>.
            </p>
          )}

          {urlLoading && (
            <p className="lede">TrustGuard AI is analyzing the URL...</p>
          )}

          {urlResult && (
            <div className="result">
              <p className="eyebrow">Risk Assessment</p>

              <h3>{urlResult.risk_level.toUpperCase()}</h3>

              <p>
                Risk score: <strong>{urlResult.risk_score}/100</strong>
              </p>

              <p>
                HTTPS:{" "}
                <strong>{urlResult.https ? "Enabled" : "Not enabled"}</strong>
              </p>

              <p>
                Hostname: <strong>{urlResult.hostname || "Unknown"}</strong>
              </p>

              <h3>URL Indicators</h3>

              {urlResult.keyword_indicators.length > 0 ? (
                <ul className="module-list">
                  {urlResult.keyword_indicators.map((item) => (
                    <li key={item}>
                      <span>{item}</span>
                      <span className="chip">Detected</span>
                    </li>
                  ))}
                </ul>
              ) : (
                <p className="lede">No suspicious keywords detected.</p>
              )}
            </div>
          )}
        </section>
      </main>

      <footer className="footer">
        Defensive research prototype. No real credentials, no offensive tooling,
        no invented accuracy metrics.
      </footer>
    </div>
  );
}
